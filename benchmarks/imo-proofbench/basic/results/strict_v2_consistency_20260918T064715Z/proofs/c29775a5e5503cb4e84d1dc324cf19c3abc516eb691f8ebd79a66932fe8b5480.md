\[ A_n = 1 + 3^{20k} + (3^2)^{14k} = 1 + 3^{20k} + 3^{28k} \]

\[ A_n = 1 + (3^{4k})^5 + (3^{4k})^7 = 1 + x^5 + x^7 \]

\[ f(\omega) = \omega^7 + \omega^5 + 1 = \omega + \omega^2 + 1 = 0 \]

\[ x^7 + x^5 + 1 = (x^2 + x + 1)(x^5 - x^4 + x^3 - x + 1) \]
